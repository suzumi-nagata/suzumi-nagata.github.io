{{- $image := .Page.Resources.GetMatch .Destination -}} {{- /* Also look in assets/ and static/, so plain Markdown images can be optimized the same way as figure shortcodes. */ -}} {{- if not $image -}} {{- $u := urls.Parse .Destination -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $path = strings.TrimPrefix "/" $path -}} {{- if not $u.IsAbs -}} {{- with resources.Get $path -}} {{- $image = . -}} {{- else -}} {{- $staticPath := printf "static/%s" $path -}} {{- if fileExists $staticPath -}} {{- $image = resources.FromString $path (os.ReadFile $staticPath) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Fallback when the image is not a page bundle resource or is a GIF/SVG. */ -}} {{- $default := printf "" (.Destination | safeURL) $.PlainText $.Title -}} {{- if $image -}} {{- $sub := lower $image.MediaType.SubType -}} {{- if in (slice "png" "jpeg" "webp") $sub -}} {{- $originalWidth := $image.Width -}} {{- /* Never upscale, and cap very large images at 1200px. */ -}} {{- $maxWidth := cond (gt $originalWidth 1200) 1200 $originalWidth -}} {{- $widths := slice -}} {{- range (slice 458 768 1200) -}} {{- if lt . $maxWidth -}} {{- $widths = $widths | append . -}} {{- end -}} {{- end -}} {{- $widths = $widths | append $maxWidth -}} {{- $srcset := slice -}} {{- $largest := false -}} {{- range $widths -}} {{- $thumb := $image.Resize (printf "%dx webp q88" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $thumb.RelPermalink .) -}} {{- $largest = $thumb -}} {{- end -}} {{- else -}} {{- $default | safeHTML -}} {{- end -}} {{- else -}} {{- warnf "Unable to find '%s' -- ensure image exists alongside document in page bundle" .Destination -}} {{- $default | safeHTML -}} {{- end -}}
